%% =============================================================================
%%  fig_filter.tex -- the snoop filter's internal mechanism, standalone.
%%
%%  Build:  pdflatex fig_filter.tex   ->  fig_filter.pdf
%%
%%  fig_arch.tex shows where the filter sits in the system. This shows what is
%%  inside it, and the one thing outside it that makes it correct.
%%
%%  Read left to right. The mirror is written from the left by the same signals
%%  that write each cache, and read from below by the address the arbiter just
%%  picked. Per (core, way) the hit decision is drawn as the RTL has it:
%%
%%      way_hit = valid[set][way]  AND  (tag_mem[set] == qry_tag)
%%
%%  -- a comparator fed by the mirror's tag read-out and the query tag, then an
%%  AND with the valid bit, then an OR across the two ways into held[c]. The
%%  other half of the decision, on the right, is who is FETCHING the line: a
%%  refill in flight is a cache that holds nothing yet and must still be told.
%%  Their union is the sharer set, and the sharer set gates the broadcast.
%%
%%  Routing rules: no signal passes through a block. Lines may cross lines --
%%  a crossing without a dot is not a connection -- but never a box. The valid
%%  box is short so the tag lines clear it; the update line to it threads the
%%  gap between the two ways and then the gap between the two comparators;
%%  qry_tag is one grey bus with a junction dot per tap; the per-core OR
%%  outputs merge on an explicit bus into the mask; gates in the bus column are
%%  rotated so that column reads top to bottom.
%%
%%  Greyscale by construction, same conventions as fig_arch: distributed RAM is
%%  the light fill, flip-flops the mid fill, the correctness term the dark one.
%% =============================================================================

\def\rowA{0} \def\rowB{-19} \def\rowC{-38} \def\rowD{-57}
\def\qbus{9.5}    % x of the qry_tag distribution bus
\def\mbus{57}     % x of the held[c] merge bus

% ------------------------------------------------------------ update ports (left)
\node[lbl,anchor=south] (updlbl) at (-42,10)
  {one update port per DCU, driven by\\the same signals that write its own arrays};
\foreach \c/\y in {0/\rowA, 1/\rowB, 2/\rowC, 3/\rowD} {
  \node[box,fill=white,minimum width=24mm,minimum height=9mm,font=\scriptsize,align=left]
    (upd\c) at (-42,\y) {\textbf{DCU \c}\\[-1pt] upd\_valid $\cdot$ set $\cdot$ way\\[-1pt] tag $\cdot$ inst};
}

% ------------------------------------------------------------- the mirror
\node[hdr,anchor=south] (mirhdr) at (-6,9.5) {Tag mirror};
\foreach \c/\y in {0/\rowA, 1/\rowB, 2/\rowC, 3/\rowD} {
  \node[ram] (t\c0) at (-6,\y+5.5) {way 0 $\cdot$ tag\_mem[64]};
  \node[ram] (t\c1) at (-6,\y-5.5) {way 1 $\cdot$ tag\_mem[64]};
  \coordinate (b\c) at (-26,\y);
  \draw[wire] (upd\c.east) -- (b\c);
  \node[dot] at (b\c) {};
  \draw[bus] (b\c) |- (t\c0.west);
  \draw[bus] (b\c) |- (t\c1.west);
}
\node[grp,fit=(t00)(t31),inner sep=2mm] (mirror) {};

% ------------------------------------------------------------- hit decision, per way
\node[hdr,anchor=south] (cmphdr) at (19,9.5) {$=$ qry\_tag};
\node[hdr,anchor=south] (vhdr)   at (29,9.5) {valid};
\node[hdr,anchor=south] (andhdr) at (41,9.5) {way\_hit};
\node[hdr,anchor=south] (orhdr)  at (52,9.5) {held[$c$]};
\foreach \c/\y in {0/\rowA, 1/\rowB, 2/\rowC, 3/\rowD} {
  % comparator per way: tag read-out from the west (upper), qry_tag tap (lower)
  \node[cmp] (e\c0) at (19,\y+5.5) {$=$};
  \node[cmp] (e\c1) at (19,\y-5.5) {$=$};
  % tag read-out straight into the comparator's centre; the qry_tag tap enters
  % the same face 1.6 mm lower, so neither line has to bend
  \draw[ln] (t\c0.east) -- (e\c0.west);
  \draw[ln] (t\c1.east) -- (e\c1.west);
  \coordinate (qt\c0) at (\qbus,\y+5.5-1.6);
  \coordinate (qt\c1) at (\qbus,\y-5.5-1.6);
  \node[dot] at (qt\c0) {}; \node[dot] at (qt\c1) {};
  \draw[ln,qwire] (qt\c0) -- (15,\y+5.5-1.6);
  \draw[ln,qwire] (qt\c1) -- (15,\y-5.5-1.6);
  % valid bits sit between the ways; their update line threads the two gaps
  \node[ff] (v\c) at (29,\y) {valid $64\!\times\!2$};
  \draw[ln] (b\c) -- (v\c.west);
  % AND per way: comparator on the outer input, valid on the inner one
  % AND gates anchored ON the input the comparator feeds, so that line is dead
  % straight; the valid bit then takes the other (inner) input
  \node[and gate US,draw,logic gate inputs=nn,scale=0.65,anchor=input 1] (a\c0) at (38.5,\y+5.5) {};
  \node[and gate US,draw,logic gate inputs=nn,scale=0.65,anchor=input 2] (a\c1) at (38.5,\y-5.5) {};
  \draw[ln] (e\c0.east) -- (a\c0.input 1);
  \draw[ln] (e\c1.east) -- (a\c1.input 2);
  \coordinate (vt\c) at (36,\y);
  \draw[wire] (v\c.east) -- (vt\c);
  \node[dot] at (vt\c) {};
  \draw[ln] (vt\c) |- (a\c0.input 2);
  \draw[ln] (vt\c) |- (a\c1.input 1);
  % OR across the two ways
  \node[or gate US,draw,logic gate inputs=nn,scale=0.75] (or\c) at (52,\y) {};
  \draw[ln] (a\c0.output) -- ++(2,0) |- (or\c.input 1);
  \draw[ln] (a\c1.output) -- ++(2,0) |- (or\c.input 2);
  % merge bus tap
  \coordinate (mt\c) at (\mbus,\y);
  \draw[wire] (or\c.output) -- (mt\c);
  \node[dot] at (mt\c) {};
}
\draw[qwire] (\qbus,-81) -- (\qbus,\rowA+5.5-1.6);
\draw[wire]  (\mbus,\rowA) -- (\mbus,\rowD);

% ------------------------------------------------------------- query path (below)
\node[box,fill=white,minimum width=30mm,minimum height=7mm,font=\scriptsize] (q) at (-10,-74)
  {\textbf{qry\_addr} = win\_addr, the arbiter's pick\\[-1pt] split into set index $\cdot$ tag};
% set index: straight up into the mirror, labelled on the line
\draw[bus] (q.north) -- (mirror.south -| q.north)
  node[tag,pos=0.5,anchor=east] {set index};
% qry_tag: out of the bottom of the query box and along a horizontal run long
% enough to carry its label, then up the bus
\draw[qwire] (q.south) -- (-10,-81) -- (\qbus,-81)
  node[tag,pos=0.5,anchor=south] {qry\_tag};

\node[lbl,anchor=north west,align=left] (stnote) at (-24,-87)
  {set index reads all eight tag arrays at once; the tag is checked in all eight comparators\\
   tags: distributed RAM --- no reset, one write port, one read port\\
   valid bits: flip-flops --- resettable, all 8 read in one cycle};

% ------------------------------------------------------------- mask requester
\node[box,fill=white,minimum width=18mm,minimum height=8mm,font=\scriptsize] (mask) at (70,-28.5)
  {clear own bit\\[-1pt] held[qry\_core] $\leftarrow$ 0};
\draw[bus] (\mbus,-28.5) -- (mask.west);
\node[dot] at (\mbus,-28.5) {};
\node[lbl,anchor=north] at (74,-33.5) {a core never invalidates\\its own copy via the bus};

\node[box,fill=white,minimum width=16mm,minimum height=7mm,font=\scriptsize\bfseries] (held) at (92,-28.5)
  {held\_o[3:0]};
\draw[bus] (mask.east) -- (held.west);

\node[grp,fit=(updlbl)(mirhdr)(cmphdr)(vhdr)(andhdr)(orhdr)(upd0)(upd3)(mirror)(q)(stnote)(e00)(e31)(or0)(or3)(mask)(held),inner sep=3.5mm,
      label={[font=\small\bfseries,inner sep=3pt]above:snoop\_filter --- an exact mirror of every cache's committed tags}] (filt) {};

% ------------------------------------------------------------- the fill term (far right)
\node[lbl,anchor=south] (filllbl) at (120,3)
  {from each DCU: is a refill of\\THIS line already in flight?};
\node[new,minimum width=34mm] (fill) at (120,-4)
  {\textbf{fills in flight}\\[-1pt]
   filling[$c$] = fill\_busy[$c$]\\[-1pt] $\wedge$ line(fill\_addr[$c$]) $=$ line(win\_addr)};

\node[or gate US,gate,logic gate inputs=nn] (uor) at (120,-20) {};
\draw[bus] (held.east) -- ++(3,0) |- ($(uor.input 1)+(0,3.5)$) -- (uor.input 1);
\draw[bus] (fill.south) -- ++(0,-2.5) -| (uor.input 2);
\node[tag,anchor=west] at (125,-19.5) {$\lor$, then $\wedge\,\neg$requester};

\node[box,fill=white,minimum width=16mm,minimum height=6.5mm,font=\scriptsize\bfseries] (sh) at (120,-33)
  {sharers[3:0]};
\draw[bus] (uor.output) -- (sh.north);

\node[or gate US,gate,logic gate inputs=nnnn] (anyor) at (120,-45) {};
\coordinate (shsplit) at (120,-38.5);
\draw[wire] (sh.south) -- (shsplit);
\foreach \i in {1,2,3,4} { \draw[ln] (shsplit) -| (anyor.input \i); }
\node[tag,anchor=west] at (125,-45) {reduce $\to$ any\_sharer};

\draw[bus] (anyor.output) -- ++(0,-2) -| (and.input 1);

\node[box,fill=white,minimum width=20mm,minimum height=7mm,font=\scriptsize\bfseries] (bn) at (120,-71)
  {bcast\_needed};
\draw[bus] (and.output) -- (bn.north);
\node[lbl,anchor=north] (bnlbl) at (120,-76)
  {0: grant now, tell nobody\\1: hold the grant until the sharers are ready};

\node[grp,fit=(filllbl)(fill)(uor)(sh)(anyor)(and)(bn)(bnlbl),inner sep=3.5mm,
      label={[font=\small\bfseries,inner sep=3pt]above:in snoopy\_bus --- residency $\cup$ imminent residency}] (busgrp) {};
